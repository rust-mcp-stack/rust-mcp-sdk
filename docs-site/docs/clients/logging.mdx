```rust
async fn handle_logging_message_notification(
    &self, params: LoggingMessageNotificationParams, _runtime: &dyn McpClient,
) -> Result<(), RpcError> {
    println!("[{}] {}: {}", params.level, params.logger.unwrap_or_default(), params.data);
    Ok(())
}
```
